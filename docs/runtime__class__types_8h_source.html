184using Slot = u4;
192struct OperandStack {
193 std::vector<Slot> stack;
194
195 // --- Push de 32 bits ---
196 void push_int(int32_t v) { stack.push_back(static_cast<u4>(v)); }
197
198 // --- Push de 64 bits (2 slots) ---
199 void push_long(int64_t v) {
200 u4 high = static_cast<u4>((v >> 32) & 0xFFFFFFFF);
201 u4 low = static_cast<u4>(v & 0xFFFFFFFF);
202 stack.push_back(high);
203 stack.push_back(low);
204 }
205
206 // --- Pop de 32 bits ---
207 int32_t pop_int() {
208 if (stack.empty())
209 throw std::runtime_error("Operand stack underflow");
210 int32_t v = static_cast<int32_t>(stack.back());
211 stack.pop_back();
212 return v;
213 }
214
215 // --- Pop de 64 bits ---
216 int64_t pop_long() {
217 if (stack.size() < 2)
218 throw std::runtime_error("Operand stack underflow");
219 u4 low = stack.back();
220 stack.pop_back();
221 u4 high = stack.back();
222 stack.pop_back();
223 return (static_cast<int64_t>(high) << 32) | low;
224 }
225
226 // --- Referências ---
227 void push_ref(RuntimeObject *ref) {
228 // armazenar ponteiro como índice ou endereço truncado (32-bit)
229 stack.push_back(static_cast<u4>(reinterpret_cast<uintptr_t>(ref)));
230 }
231
232 RuntimeObject *pop_ref() {
233 if (stack.empty())
234 throw std::runtime_error("Operand stack underflow");
235 u4 bits = stack.back();
236 stack.pop_back();
237 return reinterpret_cast<RuntimeObject *>(static_cast<uintptr_t>(bits));
238 }
239
240 // --- Float / Double helpers ---
241 void push_float(float v) {
242 union {
243 float f;
244 u4 u;
245 } conv;
246 conv.f = v;
247 stack.push_back(conv.u);
248 }
249
250 float pop_float() {
251 if (stack.empty())
252 throw std::runtime_error("Operand stack underflow");
253 union {
254 float f;
255 u4 u;
256 } conv;
257 conv.u = stack.back();
258 stack.pop_back();
259 return conv.f;
260 }
261
262 void push_double(double v) {
263 union {
264 double d;
265 u8 u;
266 } conv;
267 conv.d = v;
268 push_long(static_cast<int64_t>(conv.u));
269 }
270
271 double pop_double() {
272 union {
273 double d;
274 u8 u;
275 } conv;
276 conv.u = static_cast<u8>(pop_long());
277 return conv.d;
278 }
279
280 bool empty() const { return stack.empty(); }
281 size_t size() const { return stack.size(); }
282};

328 void push_frame(Frame *f) {
329 call_stack.push_back(f);
330
331 if (!f) {
332 DEBUG_LOG("[JVM] Push frame -> <null> | depth=" << call_stack.size());
333 return;
334 }
335
336 std::string class_name =
337 f->current_class ? f->current_class->name : "<unknown-class>";
338 std::string method_name = f->method ? f->method->name : "<unknown-method>";
339 std::string descriptor =
340 f->method ? (" " + f->method->descriptor) : std::string{};
341
342 DEBUG_LOG("[JVM] Push frame -> " << class_name << "." << method_name
343 << descriptor << " | depth=" << call_stack.size());
344 }
345 void pop_frame() {
346 if (call_stack.empty()) {
347 if (g_debug_enabled)
348 std::cerr
349 << "[JVM] Pop frame requested but call stack is already empty\n";
350 return;
351 }
352
353 Frame *top = call_stack.back();
354
355 std::string class_name =
356 top && top->current_class ? top->current_class->name : "<unknown-class>";
357 std::string method_name =
358 top && top->method ? top->method->name : "<unknown-method>";
359 std::string descriptor =
360 top && top->method ? (" " + top->method->descriptor) : std::string{};
361
362 DEBUG_LOG("[JVM] Pop frame <- " << class_name << "." << method_name
363 << descriptor << " | depth=" << (call_stack.size() - 1));
364
365 call_stack.pop_back();
366 delete top;
367 }
