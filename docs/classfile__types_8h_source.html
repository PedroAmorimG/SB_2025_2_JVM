341 std::string resolve_utf8(u2 index) const {
342 if (index == 0 || index >= constant_pool.size()) return "";
343 const auto &entry = constant_pool[index];
344
345 switch (entry.first) {
346 case ConstantTag::CONSTANT_Class:
347 return resolve_utf8(entry.second.class_info.name_index);
348 case ConstantTag::CONSTANT_Fieldref:
349 return resolve_utf8(entry.second.fieldref_info.name_and_type_index);
350 case ConstantTag::CONSTANT_Methodref:
351 return resolve_utf8(entry.second.methodref_info.name_and_type_index);
352 case ConstantTag::CONSTANT_InterfaceMethodref:
353 return resolve_utf8(entry.second.interface_methodref_info.name_and_type_index);
354 case ConstantTag::CONSTANT_NameAndType:
355 return resolve_utf8(entry.second.name_and_type_info.descriptor_index) +
356 " " + resolve_utf8(entry.second.name_and_type_info.name_index);
357 case ConstantTag::CONSTANT_Utf8: {
358 const ConstantUTF8Info &utf = entry.second.utf8_info;
359 return std::string(reinterpret_cast<const char *>(utf.bytes), utf.length);
360 }
361 default: return "";
362 }
363 return "";
364 }
